2. 判断素数
bool isPrime(int x){
    if(x < 2) 
        return false;
    for(int i=2; i*i<=x; i++){
        if(x%i==0) 
            return false;
    }
    return true;
}

6. 十进制转其他进制（通用模板）
string change(int n,int base){
    string res;
    if(n==0) 
        return "0";
    while(n){
        int t = n%base;
        if(t<10) 
            res += t+'0';
        else 
            res += t-10+'A';
        n /= base;
    }
    reverse(res.begin(),res.end());
    return res;
}
7. 统计二进制最长连续 1
int maxOne(int n){
    string s;
    int maxlen=0,len=0;
    for(char c:s){
        if(c=='1') len++;
        else len=0;
        maxlen=max(maxlen,len);
    }
    return maxlen;
}
